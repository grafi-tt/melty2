if(MELTY2_ENABLE_BENCHMARK)
  add_executable(bench_cpu bench_cpu.cc)
  target_compile_features(bench_cpu PRIVATE cxx_std_17)
  target_link_libraries(bench_cpu PRIVATE melty2)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    # the other generators are plain loops, vectorized by the compiler for the host
    target_compile_options(bench_cpu PRIVATE -march=native)
  endif()
endif()

if(NOT MELTY2_GPU_BENCHMARK STREQUAL "")
  add_executable(bench_gpu bench_gpu.cu)
  set_source_files_properties(bench_gpu.cu PROPERTIES LANGUAGE ${MELTY2_GPU_BENCHMARK})
  target_link_libraries(bench_gpu PRIVATE melty2_gpu melty2)
  set_target_properties(bench_gpu PROPERTIES ${MELTY2_GPU_BENCHMARK}_STANDARD 17 ${MELTY2_GPU_BENCHMARK}_STANDARD_REQUIRED ON)
endif()
