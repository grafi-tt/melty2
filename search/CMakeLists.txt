# The kernel search tools (see README.md):
#   cmake -B build [-DMELTY2_SEARCH_GPU=ON] [-DMELTY2_SEARCH_STEST=ON] && cmake --build build
#   MELTY2_SEARCH_CPU     build/{prof,rank,sa,weakkeys}_cpu with the CPU metric, build/check and build/state (ON)
#   MELTY2_SEARCH_GPU     build/{prof,rank,sa,weakkeys}_gpu with the HIP metric (OFF)
#   MELTY2_SEARCH_STEST   build/testu01_stest by the library testu01_gpu: an installed one (find_package), or else
#                         ../testu01_gpu built along (with its options: TestU01_ROOT, TESTU01_GPU_HOST_FLAGS) (OFF)
#   MELTY2_SEARCH_NATIVE  -march=native for the C and C++ code of the CPU (ON)
#   CMAKE_HIP_ARCHITECTURES  the GPU architectures of the HIP code (default gfx1030)
# The tools are C++17; melty2_init.c of the library is compiled as C.
cmake_minimum_required(VERSION 3.21)

option(MELTY2_SEARCH_CPU "build the tools with the CPU metric" ON)
option(MELTY2_SEARCH_GPU "build the tools with the HIP metric" OFF)
option(MELTY2_SEARCH_STEST "build testu01_stest by the library testu01_gpu" OFF)
option(MELTY2_SEARCH_NATIVE "-march=native for the code of the CPU" ON)

set(languages C CXX)
if(MELTY2_SEARCH_GPU OR MELTY2_SEARCH_STEST)
  if(NOT DEFINED CMAKE_HIP_ARCHITECTURES AND NOT DEFINED ENV{HIPARCHS})
    set(CMAKE_HIP_ARCHITECTURES gfx1030 CACHE STRING "the GPU architectures of the HIP code")
  endif()
  list(APPEND languages HIP)
endif()
project(melty2_search LANGUAGES ${languages})

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "the build type" FORCE)
endif()
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_HIP_STANDARD 17)
set(CMAKE_HIP_STANDARD_REQUIRED ON)
set(CMAKE_HIP_EXTENSIONS OFF)
# the executables in build/ itself (the scripts and the README run build/<tool>)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})

set(root ${CMAKE_CURRENT_SOURCE_DIR}/..)
set(cpu_flags $<$<BOOL:${MELTY2_SEARCH_NATIVE}>:-march=native>)

# the CPU code common to the tools: melty2_init (C); metric_common (which calls the metric) goes with each metric
add_library(melty2_search_common STATIC ${root}/src/melty2_init.c)
target_include_directories(melty2_search_common PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${root}/include)
target_compile_options(melty2_search_common PRIVATE ${cpu_flags})

set(tools prof rank sa weakkeys)

if(MELTY2_SEARCH_CPU)
  add_library(melty2_search_metric_cpu STATIC metric_cpu.cc metric_common.cc)
  target_link_libraries(melty2_search_metric_cpu PUBLIC melty2_search_common)
  target_compile_options(melty2_search_metric_cpu PRIVATE ${cpu_flags} -Wall -Wextra)
  foreach(t IN LISTS tools)
    add_executable(${t}_cpu ${t}.cc)
    target_link_libraries(${t}_cpu PRIVATE melty2_search_metric_cpu)
    target_compile_options(${t}_cpu PRIVATE ${cpu_flags} -Wall -Wextra)
  endforeach()
  add_executable(check check.cc)
  target_include_directories(check PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${root}/include)
  target_compile_options(check PRIVATE ${cpu_flags} -Wall -Wextra)
  find_package(Threads REQUIRED)
  add_executable(state state.cc)
  target_link_libraries(state PRIVATE melty2_search_common Threads::Threads)
  target_compile_options(state PRIVATE ${cpu_flags} -Wall -Wextra)
endif()

if(MELTY2_SEARCH_GPU)
  find_package(hip REQUIRED)
  add_library(melty2_search_metric_gpu STATIC metric_gpu.hip metric_common.cc)
  target_link_libraries(melty2_search_metric_gpu PUBLIC melty2_search_common hip::host)
  target_compile_options(melty2_search_metric_gpu PRIVATE $<$<COMPILE_LANGUAGE:HIP>:-Wall>
                                                          $<$<COMPILE_LANGUAGE:CXX>:${cpu_flags} -Wall -Wextra>)
  foreach(t IN LISTS tools)
    # (the tools are C++; they are linked by the HIP compiler, as the metric is HIP code)
    add_executable(${t}_gpu ${t}.cc)
    target_link_libraries(${t}_gpu PRIVATE melty2_search_metric_gpu)
    target_compile_options(${t}_gpu PRIVATE ${cpu_flags} -Wall -Wextra)
    set_target_properties(${t}_gpu PROPERTIES LINKER_LANGUAGE HIP)
  endforeach()
endif()

if(MELTY2_SEARCH_STEST)
  find_package(hip REQUIRED)
  if(NOT TARGET testu01_gpu::testu01_gpu)
    find_package(testu01_gpu 0.1 CONFIG QUIET)
  endif()
  if(NOT TARGET testu01_gpu::testu01_gpu)
    set(TESTU01_GPU_EXAMPLES OFF CACHE BOOL "build the example splitmix_bigcrush")
    add_subdirectory(${root}/testu01_gpu ${CMAKE_BINARY_DIR}/testu01_gpu)
  endif()
  add_executable(testu01_stest testu01_stest.hip)
  target_link_libraries(testu01_stest PRIVATE melty2_search_common testu01_gpu::testu01_gpu hip::host)
  target_compile_options(testu01_stest PRIVATE -Wall)
endif()
