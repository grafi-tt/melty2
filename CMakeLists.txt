cmake_minimum_required(VERSION 3.10)
project(melty2 LANGUAGES C CXX)

option(BUILD_SHARED_LIBS "Build library as shared" OFF)
option(MELTY2_ENABLE_CPU_FEATURES "Enable runtime CPU detection by cpu_features" ON)
option(MELTY2_ENABLE_TEST "Enable test targets" ON)
option(MELTY2_ENABLE_BIGCRUSH "Enable bigcrush targets" OFF)
option(MELTY2_ENABLE_BENCHMARK "Enable the CPU benchmark target" OFF)
set(MELTY2_GPU_TEST "" CACHE STRING "Build the GPU test by CUDA or HIP (empty to disable)")
set(MELTY2_GPU_BENCHMARK "" CACHE STRING "Build the GPU benchmark by CUDA or HIP (empty to disable)")
option(MELTY2_INTERNAL_SWAP32_CTR "Swap 32bits of counter for testing (internal use)" OFF)

if(MELTY2_ENABLE_CPU_FEATURES)
  # Build cpu_featurs as static library
  if(DEFINED ENABLE_INSTALL)
    set(${PROJECT_NAME}_ENABLE_INSTALL_BAK ${ENABLE_INSTALL})
  endif()
  set(ENABLE_INSTALL ON CACHE INTERNAL "" FORCE)
  set(${PROJECT_NAME}_BUILD_SHARED_LIBS_BAK ${BUILD_SHARED_LIBS})
  set(BUILD_SHARED_LIBS OFF CACHE INTERNAL "" FORCE)
  add_subdirectory(third_party/cpu_features EXCLUDE_FROM_ALL)
  set(BUILD_SHARED_LIBS ${${PROJECT_NAME}_BUILD_SHARED_LIBS_BAK} CACHE INTERNAL "" FORCE)
  unset(${PROJECT_NAME}_BUILD_SHARED_LIBS_BAK)
  if(DEFINED ${PROJECT_NAME}_ENABLE_INSTALL_BAK)
    set(ENABLE_INSTALL ${${PROJECT_NAME}_ENABLE_INSTALL_BAK} CACHE INTERNAL "" FORCE)
    unset(${PROJECT_NAME}_ENABLE_INSTALL_BAK)
  endif()
endif()

add_subdirectory(src)

# Enable the language named by the variable var for GPU (CUDA, HIP or empty)
macro(melty2_enable_gpu_language_ var)
  if(${var} STREQUAL "CUDA" OR ${var} STREQUAL "HIP")
    if(CMAKE_VERSION VERSION_LESS 3.21)
      message(FATAL_ERROR "${var} requires CMake 3.21 or later")
    endif()
    enable_language(${${var}})
  elseif(NOT ${var} STREQUAL "")
    message(FATAL_ERROR "${var} must be CUDA, HIP or empty")
  endif()
endmacro()

if(MELTY2_ENABLE_TEST)
  melty2_enable_gpu_language_(MELTY2_GPU_TEST)
  enable_testing()
  add_subdirectory(test)
endif()
if(MELTY2_ENABLE_BIGCRUSH)
  add_subdirectory(bigcrush)
endif()
if(MELTY2_ENABLE_BENCHMARK OR NOT MELTY2_GPU_BENCHMARK STREQUAL "")
  melty2_enable_gpu_language_(MELTY2_GPU_BENCHMARK)
  add_subdirectory(bench)
endif()
